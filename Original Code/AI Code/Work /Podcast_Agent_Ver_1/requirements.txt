# Agentic Podcast Summariser — MCP + Gemini + Prefab
#

fastmcp[apps]        # MCP server framework; [apps] pulls in prefab-ui (Prefab UI)

yt-dlp               # YouTube audio download (RSS fallback source)
feedparser           # RSS parsing
mutagen              # Audio duration probing
requests             # Direct audio-enclosure download
PyYAML               # podcasts.yaml
python-dotenv        # .env loading
